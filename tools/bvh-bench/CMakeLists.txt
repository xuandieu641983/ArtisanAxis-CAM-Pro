cmake_minimum_required(VERSION 3.22)
project(ArtisanBVHBenchmark LANGUAGES CXX)
get_filename_component(ARTISAN_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
find_package(Threads REQUIRED)
add_executable(ArtisanBVHBenchmark benchmark.cpp "${ARTISAN_SOURCE_DIR}/cam/geometry.cpp")
target_compile_features(ArtisanBVHBenchmark PRIVATE cxx_std_20)
target_include_directories(ArtisanBVHBenchmark PRIVATE "${ARTISAN_SOURCE_DIR}")
target_link_libraries(ArtisanBVHBenchmark PRIVATE Qt6::Core Threads::Threads)
if(WIN32)
    target_link_libraries(ArtisanBVHBenchmark PRIVATE Psapi)
endif()
if(MSVC)
    target_compile_options(ArtisanBVHBenchmark PRIVATE /utf-8)
endif()
