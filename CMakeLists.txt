cmake_minimum_required(VERSION 3.22)
project(ArtisanAxis VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Widgets Gui Concurrent OpenGL OpenGLWidgets)
add_executable(ArtisanAxis WIN32 main.cpp view.cpp stock.cpp profiles.cpp workflow.cpp)
target_compile_definitions(ArtisanAxis PRIVATE ARTISAN_VERSION="${PROJECT_VERSION}")
target_link_libraries(ArtisanAxis PRIVATE Qt6::Widgets Qt6::Concurrent Qt6::OpenGL Qt6::OpenGLWidgets)
find_package(Threads REQUIRED)
add_library(ArtisanGeometry STATIC cam/geometry.cpp cam/wrap.cpp cam/planner.cpp cam/nc.cpp cam/motion.cpp cam/nc_motion.cpp cam/stock_sim.cpp)
target_include_directories(ArtisanGeometry PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(ArtisanGeometry PUBLIC Threads::Threads)
add_executable(CAMWorker worker.cpp cam/job.cpp cam/rhino.cpp)
option(ARTISAN_RHINO "Enable static openNURBS .3dm import" OFF)
if(ARTISAN_RHINO)
    include(FetchContent)
    FetchContent_Declare(artisan_opennurbs GIT_REPOSITORY https://github.com/mcneel/opennurbs.git
                         GIT_TAG eb92af3ba1806b0a34a99aba0d3bda83e3d46083)
    FetchContent_GetProperties(artisan_opennurbs)
    if(NOT artisan_opennurbs_POPULATED)
        FetchContent_Populate(artisan_opennurbs)
        add_subdirectory(${artisan_opennurbs_SOURCE_DIR} ${artisan_opennurbs_BINARY_DIR} EXCLUDE_FROM_ALL)
    endif()
    target_include_directories(CAMWorker SYSTEM PRIVATE ${artisan_opennurbs_SOURCE_DIR})
    target_compile_definitions(CAMWorker PRIVATE ARTISAN_RHINO)
    install(FILES ${artisan_opennurbs_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME openNURBS.txt)
    target_link_libraries(CAMWorker PRIVATE opennurbsStatic)
    # The bundled zlib refers back to the allocator in the static Rhino archive.
    if(UNIX AND NOT APPLE)
        target_link_options(CAMWorker PRIVATE "LINKER:--undefined=zcalloc")
    endif()
endif()
target_link_libraries(CAMWorker PRIVATE Qt6::Core ArtisanGeometry)
foreach(target ArtisanAxis CAMWorker ArtisanGeometry)
    if(MSVC)
        target_compile_options(${target} PRIVATE /utf-8 /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
option(ARTISAN_DISCOVER_CAM_SDK "Check dependencies for subsequent geometry milestones" OFF)
if(ARTISAN_DISCOVER_CAM_SDK)
    include(cmake/CamSdk.cmake)
endif()
install(TARGETS ArtisanAxis CAMWorker RUNTIME DESTINATION .)
install(FILES scripts/Update.ps1 scripts/Install-Update.ps1 scripts/update-feed.json DESTINATION scripts)
include(CTest)
if(BUILD_TESTING)
    add_executable(ArtisanChecks tests/checks.cpp stock.cpp profiles.cpp workflow.cpp)
    target_include_directories(ArtisanChecks PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(ArtisanChecks PRIVATE Qt6::Widgets)
    if(MSVC)
        target_compile_options(ArtisanChecks PRIVATE /utf-8)
    endif()
    add_test(NAME ConfigAndStock COMMAND ArtisanChecks)
    set_tests_properties(ConfigAndStock PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
if(BUILD_TESTING)
    add_executable(GeometryChecks tests/geometry_checks.cpp)
    target_link_libraries(GeometryChecks PRIVATE ArtisanGeometry)
    add_test(NAME GeometryAnd4XPost COMMAND GeometryChecks)
    add_test(NAME Worker4XIntegration COMMAND ${CMAKE_COMMAND} "-DWORKER=$<TARGET_FILE:CAMWorker>"
             "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/worker.cmake")
endif()
if(BUILD_TESTING)
    add_executable(MotionChecksTest tests/motion_checks.cpp)
    target_link_libraries(MotionChecksTest PRIVATE ArtisanGeometry)
    add_test(NAME SimultaneousMotionAndNc COMMAND MotionChecksTest)
endif()
if(BUILD_TESTING)
    add_test(NAME WorkerSimultaneousFile COMMAND ${CMAKE_COMMAND} "-DWORKER=$<TARGET_FILE:CAMWorker>"
             "-DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/worker_motion.cmake")
endif()
if(BUILD_TESTING)
    add_executable(DialogAudit tests/dialog_audit.cpp stock.cpp profiles.cpp workflow.cpp)
    target_include_directories(DialogAudit PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(DialogAudit PRIVATE Qt6::Widgets)
    add_executable(WorkshopAudit tests/workshop_audit.cpp view.cpp stock.cpp profiles.cpp workflow.cpp)
    add_dependencies(WorkshopAudit CAMWorker)
    target_link_libraries(WorkshopAudit PRIVATE Qt6::Widgets Qt6::OpenGL Qt6::OpenGLWidgets)
    if(MSVC)
        target_compile_options(DialogAudit PRIVATE /utf-8)
        target_compile_options(WorkshopAudit PRIVATE /utf-8)
    endif()
    add_test(NAME DialogAudit COMMAND DialogAudit)
    add_test(NAME WorkshopAudit COMMAND WorkshopAudit "${CMAKE_CURRENT_SOURCE_DIR}/tests/spiral_cylinder.stl")
    set_tests_properties(DialogAudit WorkshopAudit PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
endif()
if(BUILD_TESTING)
    add_executable(FinishToleranceCheck tests/finish_tolerance.cpp)
    target_link_libraries(FinishToleranceCheck PRIVATE ArtisanGeometry)
    add_test(NAME FinishTolerance COMMAND FinishToleranceCheck)
endif()
